pragma solidity >=0.7.0 <0.9.0;

contract SimpleBank {
    mapping(address => uint) private balances;
    address public owner;

    // Constructor
    constructor() payable {
        owner = msg.sender;
    }

    // Deposit Ether into the bank
    function deposit() public payable returns (uint) {
        require(msg.value > 0, "Deposit must be greater than zero");
        balances[msg.sender] += msg.value;
        return balances[msg.sender];
    }

    // Withdraw Ether from the bank
    function withdraw(uint withdrawAmount) public returns (uint) {
        require(withdrawAmount > 0, "Amount must be greater than zero");
        require(withdrawAmount <= balances[msg.sender], "Insufficient balance");

        balances[msg.sender] -= withdrawAmount;
        (bool success, ) = payable(msg.sender).call{value: withdrawAmount}("");
        require(success, "Transfer failed");

        return balances[msg.sender];
    }

    // Show the balance of the caller
    function balance() public view returns (uint) {
        return balances[msg.sender];
    }
}
